-- Thực hiện riêng từng câu lệnh trong Microsoft Access (SQL View).
-- Sao lưu cuahang.dll trước khi thực hiện. Tệp này thực chất là cơ sở dữ liệu Access.
-- Nếu NGUOI_DUNG đã tồn tại đúng cấu trúc thì KHÔNG tạo lại bảng.

CREATE TABLE NGUOI_DUNG (
 ID COUNTER CONSTRAINT PK_NGUOI_DUNG PRIMARY KEY,
 TEN_DANG_NHAP TEXT(100) NOT NULL,
 MATKHAU TEXT(255) NOT NULL,
 SALT TEXT(255),
 ROLE TEXT(20) NOT NULL,
 HOTEN TEXT(150),
 IS_ACTIVE YESNO,
 MUST_CHANGE_PWD YESNO,
 CREATED_AT DATETIME,
 UPDATED_AT DATETIME
);

CREATE UNIQUE INDEX UX_NGUOI_DUNG_TEN_DANG_NHAP ON NGUOI_DUNG (TEN_DANG_NHAP);
